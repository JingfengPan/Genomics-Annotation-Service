#!/bin/bash -ex

# Download gas.zip from S3
aws s3 cp s3://mpcs-students/jfpan01/gas.zip /tmp/gas.zip

# Unzip to /home/ubuntu/gas
cd /home/ubuntu
unzip -q /tmp/gas.zip -d /home/ubuntu

# Change ownership to ubuntu:ubuntu for all GAS files
chown -R ubuntu:ubuntu /home/ubuntu/gas

# Make run_gas.sh executable
chmod +x /home/ubuntu/gas/web/run_gas.sh

# Run the web server as ubuntu user
sudo -u ubuntu bash -c "cd /home/ubuntu/gas/web && /home/ubuntu/gas/web/run_gas.sh"
